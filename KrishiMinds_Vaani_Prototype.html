System Overview
Full Technical Architecture
Every layer of the VAANI V3 system — what it is, what it does, what tech stack it uses, and how it connects to everything else.
Layer 1
Data Ingestion
Weather API, NDVI satellite, pest bulletins, crop calendars, Syngenta CRM, farmer pulse reports. Apache Airflow schedules every 6 hours.
→
Layer 2
Intelligence Engine
Territory scoring, emotion classifier, content generation (LLM), visit routing (TSP), anomaly detection, champion rewards engine.
→
Layer 3
API Layer
FastAPI REST backend. Endpoints for rep app, dealer WhatsApp webhooks, farmer pulse responses, management dashboard. JWT auth.
→
Layer 4
Delivery Layer
Rep PWA app (React + SQLite offline). WhatsApp Business API (Twilio). IVR calls (Exotel). SMS (MSG91). Management dashboard.
→
Layer 5
Learning Loop
Visit outcomes → memory DB update. Pulse reports → territory score recalibration. Champion referrals → rewards engine. Weekly model retrain.
Backend
Python + FastAPI
REST API server. Handles all business logic. Connects to DB, LLM, messaging services. Runs on AWS EC2 or Railway.
Database
PostgreSQL + Redis
PostgreSQL for all persistent data — farmers, memory, visits, scores. Redis for caching hot data and task queues.
LLM
Claude API + LangChain
Content generation engine. LangChain orchestrates prompts. IndicTrans2 for Indian language translation. Runs per-request.
Frontend — Rep App
React + PWA + SQLite
Progressive Web App. Offline-first using service workers + SQLite (via sql.js). Pre-downloads daily plan. Syncs when online.
Messaging
Twilio + Exotel + MSG91
Twilio: WhatsApp Business API. Exotel: IVR outbound calls (India). MSG91: SMS gateway. All via REST API calls from backend.
Data Pipeline
Apache Airflow
Schedules and orchestrates all data ingestion tasks. Weather every 6h, pest bulletins daily, NDVI weekly, model retrain weekly.

@router.get("/daily-plan/{rep_id}")
async def get_daily_plan(
    rep_id: int,
    current_rep=Depends(get_current_rep),
    db=Depends(get_db)
):
    """
    Returns rep's full day:
    - Territory-scoped stops only (rep's assigned districts)
    - All 5 stop types: DEALER, FARMER, CHAMPION, INSTITUTIONAL
    - Each stop has: memory brief, talking points, emotion mode
    - Optimized route order (TSP + urgency weights)

    return plan

2.3 — Farmer Pulse Endpoint
backend/api/farmers.pypython
@router.post("/pulse-response")
async def receive_pulse_response(
    farmer_phone: str,
    response: int,  # 1=healthy, 2=mild, 3=severe
    db=Depends(get_db)
):
    """
    Called when farmer replies to WhatsApp check-in or presses IVR key.
    Updates: farmer memory + territory pulse score + triggers actions.
    """
    farmer = get_farmer_by_phone(farmer_phone, db)

    # Update memory record
    update_memory(farmer.id, {
        "last_pulse_response": response,
        "last_pulse_date": datetime.now(),
        "last_reported_condition": ["healthy", "mild", "severe"][response-1]
    }, db)

    # Update territory pulse score
    update_pulse_count(farmer.district_id, response, db)

    # Auto-trigger actions based on response
    if response >= 2:
        # Brief the nearest dealer
        trigger_dealer_brief(farmer.nearest_dealer_id, farmer, db)
    if response == 3:
        # Emergency — add farmer field visit to rep's plan
        add_urgent_visit(farmer, db)
        # Alert management dashboard
        send_emergency_alert(farmer, db)

    # Send follow-up message to farmer
    followup = generate_followup_message(farmer, response)
    send_whatsapp(farmer.phone, followup)

    return {"status": "processed"}

    Core content generation endpoint.
    Called by: Airflow daily (bulk), or manually (demo).
    Returns: ready-to-send message for specified tier + language.

    context = {

        "crop": request.crop,
        "stage": request.stage,
        "pest": get_active_pest(request.district_id, db),
        "territory_score": get_territory_score(request.district_id, db),
        "emotional_mode": get_emotional_mode(request.district_id, db),

        "memory": get_farmer_memory(request.farmer_id, db) if request.farmer_id else None,
        "voice_persona": request.voice_persona,
        "language": request.language,
        "trust_level": request.trust_level,
    }

    # 2. Generate with LLM

    # 3. Translate to target language
    translated = translate_to_language(content, request.language)

    # 4. Format for delivery channel
    formatted = format_for_channel(translated, request.channel)

    return {"content": formatted, "context_used": context}

from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta
import requests, os

DISTRICTS = [
    {"name": "Krishna", "lat": 16.57, "lon": 80.96},
    {"name": "Guntur",  "lat": 16.30, "lon": 80.43},
    # ... all 20 districts
]

def fetch_weather_for_all_districts():
    api_key = os.getenv("OPENWEATHERMAP_API_KEY")
    results = []
    for d in DISTRICTS:
        url = f"https://api.openweathermap.org/data/2.5/weather?lat={d['lat']}&lon={d['lon']}&appid={api_key}"
        r = requests.get(url).json()
        results.append({
            "district": d["name"],
            "temp": r["main"]["temp"] - 273.15,
            "humidity": r["main"]["humidity"],
            "rainfall": r.get("rain", {}).get("1h", 0),
            "timestamp": datetime.now().isoformat()
        })
    # Store in PostgreSQL
    save_weather_data(results)
    # Trigger territory score recalculation
    recalculate_all_territory_scores()

with DAG(
    "vaani_weather_pipeline",
    schedule_interval="0 */6 * * *",  # Every 6 hours
    start_date=datetime(2026, 5, 13),
    catchup=False
) as dag:

    fetch_task = PythonOperator(
        task_id="fetch_weather",
        python_callable=fetch_weather_for_all_districts

4.3 — Territory Score Calculation (The Core Logic)
backend/services/territory_score.pypython
def calculate_territory_score(district_id: int, db) -> dict:
    """
    Composite score from 5 signals.
    Returns score 0-100 + urgency level + emotional mode.
    """
    district = db.query(District).filter_by(id=district_id).first()

    # Signal 1: Pest Risk (35% weight)
    pest_score = get_active_pest_severity(district_id, db) * 0.35

    # Signal 2: Weather Risk (25% weight)
    weather = get_latest_weather(district_id, db)
    humidity_risk = min(weather.humidity / 100, 1.0)
    rainfall_deviation = get_rainfall_deviation(district_id, db)
    weather_score = (humidity_risk * 0.5 + rainfall_deviation * 0.5) * 25

    # Signal 3: Crop Stage Urgency (25% weight)
    stage = get_current_crop_stage(district_id, db)
    STAGE_WEIGHTS = {"Tillering": 0.9, "Flowering": 1.0, "Vegetative": 0.6}
    stage_score = STAGE_WEIGHTS.get(stage, 0.5) * 25

    # Signal 4: Farmer Pulse Reports (15% weight) ★ V3 Addition
    recent_reports = get_pulse_reports_48h(district_id, db)
    severe_reports = [r for r in recent_reports if r.response == 3]
    pulse_score = min(len(severe_reports) / 10, 1.0) * 15

    composite = pest_score + weather_score + stage_score + pulse_score

    # Determine urgency
    urgency = ("CRITICAL" if composite >= 80
               else "HIGH" if composite >= 65
               else "MEDIUM" if composite >= 40
               else "LOW")

    # ★ V3: Determine emotional mode
    panic_count = district.panic_report_count
    emotional_mode = ("panic"     if panic_count >= 5
                     else "recover"  if composite > 50 and composite < 65
                     else "confident" if composite < 30
                     else "stable")

    return {"score": round(composite, 1), "urgency": urgency, "mode": emotional_mode}

def generate_with_llm(context: dict) -> str:
    voice = context["voice_persona"]
    mode = context["emotional_mode"]
    memory = context.get("memory", {})
    trust = context.get("trust_level", "cold")

    # Build memory reference if available
    memory_ref = ""
    if memory and trust in ["warm", "trusted"]:
        memory_ref = f"Previous interaction: {memory.get('last_problem', '')}. Outcome: {memory.get('last_outcome', '')}. Reference this naturally."

    system_prompt = f"""You are generating agricultural communication content for VAANI.
You are speaking as: {VOICE_PERSONAS.get(voice, VOICE_PERSONAS['vaani'])}

EMOTIONAL MODE RULE: {EMOTIONAL_MODE_RULES.get(mode, EMOTIONAL_MODE_RULES['stable'])}

TRUST LEVEL: {trust}
- cold: NO product mention. Pure value and care only.
- warm: Soft product reference acceptable if highly relevant.
- trusted: Natural product recommendation appropriate.

{f'MEMORY CONTEXT: {memory_ref}' if memory_ref else ''}

CONTENT RULES:
1. Open with a hook that creates curiosity — never with product name
2. Slight controlled imperfection matches the voice persona
3. Keep it conversational — never formal corporate language
4. Product mention only after establishing genuine relevance
5. End with care, not a call-to-action"""

    user_prompt = f"""Generate content for:
District: {context['district']}
Crop: {context['crop']} | Stage: {context['stage']}
Active threat: {context.get('pest', 'general monitoring')}
Territory score: {context.get('territory_score', 50)}/100
Weather: {context.get('weather', {}).get('condition', 'normal')}
Language: Generate in English first (will be translated)

Generate the message now. 80-150 words maximum."""

    message = client.messages.create(
        model="claude-sonnet-4-20250514",
        max_tokens=400,
        system=system_prompt,
        messages=[{"role": "user", "content": user_prompt}]
    )
    return message.content[0].text
5.2 — Translation Pipeline (English → Indian Languages)
backend/services/translation.pypython
from google.cloud import translate_v2 as translate
import os

LANGUAGE_CODES = {
    "Telugu": "te", "Hindi": "hi", "Tamil": "ta",
    "Marathi": "mr", "Kannada": "kn", "Bengali": "bn"
}

# Agricultural terms that should NOT be translated
# (farmers know brand names in English)
KEEP_IN_ENGLISH = [
    "Actara 25WG", "Engeo Pleno", "Amistar Top",
    "Karate Zeon", "Coragen 20 SC", "VAANI"
]

translate_client = translate.Client()

def translate_to_language(text: str, language: str) -> str:
    if language == "English":
        return text

    lang_code = LANGUAGE_CODES.get(language, "hi")

    # Protect brand names with placeholders before translation
    protected = text
    placeholders = {}
    for i, term in enumerate(KEEP_IN_ENGLISH):
        placeholder = f"BRAND{i}"
        if term in protected:
            placeholders[placeholder] = term
            protected = protected.replace(term, placeholder)

    # Translate
    result = translate_client.translate(protected, target_language=lang_code)
    translated = result["translatedText"]

    # Restore brand names
    for placeholder, term in placeholders.items():
        translated = translated.replace(placeholder, term)

    return translated

import initSqlJs from 'sql.js';

let db = null;

export async function initDB() {
  const SQL = await initSqlJs({
    locateFile: file => `/sql-wasm.wasm`
  });
  db = new SQL.Database();

  // Create tables for offline storage
  db.run(`
    CREATE TABLE IF NOT EXISTS daily_plan (
      id INTEGER PRIMARY KEY,
      data TEXT,           -- Full JSON plan
      downloaded_at TEXT,
      date TEXT
    );
    CREATE TABLE IF NOT EXISTS pending_logs (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      stop_id INTEGER,
      stop_type TEXT,
      outcome TEXT,        -- JSON outcome data
      logged_at TEXT,
      synced INTEGER DEFAULT 0
    );
    CREATE TABLE IF NOT EXISTS farmer_memory_cache (
      farmer_id INTEGER PRIMARY KEY,
      memory_data TEXT,    -- JSON memory record
      cached_at TEXT
    );
  `);
}

export function saveDailyPlan(planData) {
  // Store today's plan locally
  db.run(`INSERT OR REPLACE INTO daily_plan VALUES (1, ?, ?, ?)`,
    [JSON.stringify(planData), new Date().toISOString(), planData.date]);
}

export function getDailyPlan() {
  const result = db.exec(`SELECT data FROM daily_plan WHERE id = 1`);
  if (!result[0]) return null;
  return JSON.parse(result[0].values[0][0]);
}

export function logVisitOffline(stopId, stopType, outcome) {
  // Queue outcome log — will sync when online
  db.run(`INSERT INTO pending_logs (stop_id, stop_type, outcome, logged_at) VALUES (?, ?, ?, ?)`,
    [stopId, stopType, JSON.stringify(outcome), new Date().toISOString()]);
}

export function getPendingLogs() {
  const result = db.exec(`SELECT * FROM pending_logs WHERE synced = 0`);
  return result[0]?.values || [];
}

export function markLogsSynced() {
  db.run(`UPDATE pending_logs SET synced = 1 WHERE synced = 0`);
}

import axios from 'axios';
import { saveDailyPlan, getPendingLogs, markLogsSynced } from '../db/offlineDb';

const API = process.env.VITE_API_URL;

export async function morningSync(repId) {
  // Step 1: Upload any queued outcome logs from yesterday
  const pending = getPendingLogs();
  if (pending.length > 0) {
    await axios.post(`${API}/api/reps/sync-logs`, { logs: pending });
    markLogsSynced();
    console.log(`✅ Synced ${pending.length} pending logs`);

  // Step 2: Download today's daily plan

  return plan.data;
}

import requests

def make_ivr_call(phone_number: str, script: str, language: str):
    """
    Makes outbound IVR call to farmer's phone.
    Works on any phone — no smartphone / WhatsApp needed.
    Farmer hears voice message and presses 1/2/3.
    Keypress captured and sent to /webhooks/exotel-response
    """
    # Step 1: Convert text to speech audio file
    audio_url = text_to_speech(script, language)

    # Step 2: Make call via Exotel
    subdomain = os.getenv("EXOTEL_SUBDOMAIN")
    url = f"https://{subdomain}.exotel.com/v1/Accounts/{subdomain}/Calls/connect"

    response = requests.post(url,
        auth=(os.getenv("EXOTEL_API_KEY"), os.getenv("EXOTEL_API_TOKEN")),
        data={
            "From": phone_number,
            "To": "EXOTEL_CALLER_ID",
            "Url": audio_url,
            "StatusCallback": "https://yourapp.com/webhooks/exotel-response"
        }
    )
    return response.json()


def text_to_speech(text: str, language: str) -> str:
    """Convert text to audio file, return URL."""
    from google.cloud import texttospeech

    LANGUAGE_VOICES = {
        "Telugu":  ("te-IN", "te-IN-Standard-A"),
        "Hindi":   ("hi-IN", "hi-IN-Standard-A"),
        "Tamil":   ("ta-IN", "ta-IN-Standard-A"),
        "Marathi": ("mr-IN", "mr-IN-Standard-A"),
    }
    lang_code, voice_name = LANGUAGE_VOICES.get(language, ("hi-IN", "hi-IN-Standard-A"))

    tts_client = texttospeech.TextToSpeechClient()
    synthesis_input = texttospeech.SynthesisInput(text=text)
    voice = texttospeech.VoiceSelectionParams(language_code=lang_code, name=voice_name)
    audio_config = texttospeech.AudioConfig(audio_encoding=texttospeech.AudioEncoding.MP3)

    response = tts_client.synthesize_speech(
        input=synthesis_input, voice=voice, audio_config=audio_config
    )
    # Save to S3/GCS and return URL
    audio_url = save_audio_to_storage(response.audio_content)
    return audio_url
